=LAMBDA(txt,
  LET(
    asc, ASCII_(txt),
    IF(asc = "",
  LET(
    modBig,  4294967296,
    maskAll, 4294967295,

    addM, LAMBDA(xx, yy, MOD(xx + yy, modBig)),
    notM, LAMBDA(xx, BITXOR(xx, maskAll)),
    rotL, LAMBDA(xx, nn,
      MOD(
        BITOR(
          BITLSHIFT(BITAND(xx, 2^(32 - nn) - 1), nn),
          BITRSHIFT(xx, 32 - nn)
        ),
        modBig
      )
    ),

    mlen, LEN(txt),
    plen, 64 * (INT((mlen + 8) / 64) + 1),
    nblk, plen / 64,

    padByte, LAMBDA(idx,
      IF(idx < mlen,
        CODE(MID(txt, idx + 1, 1)),
        IF(idx = mlen,
          128,
          IF(idx >= plen - 8,
            MOD(INT(mlen * 8 / 256^(idx - plen + 8)), 256),
            0
          )
        )
      )
    ),

    getWord, LAMBDA(blkIdx, wordIdx,
      LET(
        pp, blkIdx * 64 + wordIdx * 4,
        padByte(pp)
          + padByte(pp + 1) * 256
          + padByte(pp + 2) * 65536
          + padByte(pp + 3) * 16777216
      )
    ),

    karr, INT(modBig * ABS(SIN(SEQUENCE(64)))),

    shiftV, LAMBDA(idx,
      LET(
        rr, MOD(idx, 4),
        qq, INT(idx / 16),
        IF(qq = 0, CHOOSE(rr + 1, 7, 12, 17, 22),
        IF(qq = 1, CHOOSE(rr + 1, 5,  9, 14, 20),
        IF(qq = 2, CHOOSE(rr + 1, 4, 11, 16, 23),
                   CHOOSE(rr + 1, 6, 10, 15, 21))))
      )
    ),

    hInit, CHOOSE({1, 2, 3, 4}, 1732584193, 4023233417, 2562383102, 271733878),

    finalH, REDUCE(hInit, SEQUENCE(nblk, , 0),
      LAMBDA(hh, blk,
        LET(
          rounds, REDUCE(hh, SEQUENCE(64, , 0),
            LAMBDA(state, ii,
              LET(
                aa, INDEX(state, 1, 1),
                bb, INDEX(state, 1, 2),
                cc, INDEX(state, 1, 3),
                dd, INDEX(state, 1, 4),
                qq, INT(ii / 16),
                ff, IF(qq = 0, BITOR(BITAND(bb, cc),       BITAND(notM(bb), dd)),
                    IF(qq = 1, BITOR(BITAND(dd, bb),       BITAND(notM(dd), cc)),
                    IF(qq = 2, BITXOR(BITXOR(bb, cc), dd),
                               BITXOR(cc, BITOR(bb, notM(dd)))))),
                gg, IF(qq = 0, ii,
                    IF(qq = 1, MOD(5 * ii + 1, 16),
                    IF(qq = 2, MOD(3 * ii + 5, 16),
                               MOD(7 * ii, 16)))),
                tmp,  addM(addM(addM(aa, ff), INDEX(karr, ii + 1)), getWord(blk, gg)),
                newB, addM(bb, rotL(tmp, shiftV(ii))),
                CHOOSE({1, 2, 3, 4}, dd, newB, bb, cc)
              )
            )
          ),
          CHOOSE({1, 2, 3, 4},
            addM(INDEX(hh, 1, 1), INDEX(rounds, 1, 1)),
            addM(INDEX(hh, 1, 2), INDEX(rounds, 1, 2)),
            addM(INDEX(hh, 1, 3), INDEX(rounds, 1, 3)),
            addM(INDEX(hh, 1, 4), INDEX(rounds, 1, 4))
          )
        )
      )
    ),

    hexByte, LAMBDA(bv, RIGHT("0" & DEC2HEX(bv), 2)),
    wordHex, LAMBDA(ww,
        hexByte(MOD(ww, 256))
      & hexByte(MOD(INT(ww / 256),      256))
      & hexByte(MOD(INT(ww / 65536),    256))
      & hexByte(MOD(INT(ww / 16777216), 256))
    ),

    LOWER(
        wordHex(INDEX(finalH, 1, 1))
      & wordHex(INDEX(finalH, 1, 2))
      & wordHex(INDEX(finalH, 1, 3))
      & wordHex(INDEX(finalH, 1, 4))
    )
  ),
      asc
    )
  )
)
