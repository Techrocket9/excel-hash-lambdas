=LAMBDA(tx,
  LET(
    n,LEN(tx),
    p,64*INT(n/64+9/8),
    i,SEQUENCE(p,,0),
    u,IFERROR(UNICODE(MID(tx,i+1,1)),999),
    M,2^32,
    kc,INT(M*ABS(SIN(SEQUENCE(64)))),
    ro,LAMBDA(xv,kv,
      MOD(xv,2^(32-kv))*2^kv+INT(xv/2^(32-kv))
    ),
    wd,MMULT(WRAPROWS(IF(i<n,u,IF(i=n,128,IF(i<p-8,0,MOD(INT(8*n/256^(i-p+8)),256)))),4),256^{0;1;2;3}),
    fh,REDUCE({1732584193,4023233417,2562383102,271733878},SEQUENCE(p/64,,0),LAMBDA(hs,bk,
      LET(
        ws,TAKE(DROP(wd,16*bk),16),
        MOD(hs+REDUCE(hs,SEQUENCE(64,,0),LAMBDA(s,j,
          LET(
            bb,INDEX(s,2),
            cc,INDEX(s,3),
            dd,INDEX(s,4),
            qq,INT(j/16),
            ff,CHOOSE(qq+1,BITXOR(dd,BITAND(bb,BITXOR(cc,dd))),BITXOR(cc,BITAND(dd,BITXOR(bb,cc))),BITXOR(BITXOR(bb,cc),dd),BITXOR(cc,BITOR(bb,M-1-dd))),
            gg,CHOOSE(qq+1,j,MOD(5*j+1,16),MOD(3*j+5,16),MOD(7*j,16)),
            MOD(CHOOSECOLS(s,4,2,2,3)+{0,1,0,0}*ro(MOD(INDEX(s,1)+ff+INDEX(kc,j+1)+INDEX(ws,gg+1),M),INDEX({7,12,17,22;5,9,14,20;4,11,16,23;6,10,15,21},qq+1,MOD(j,4)+1)),M)
          )
        )),M)
      )
    )),
    IF(OR((i<n)*(u>127)),"Error: non-ASCII input detected",LOWER(CONCAT(DEC2HEX(TOCOL(MOD(INT(fh/256^{0;1;2;3}),256),,TRUE),2))))
  )
)
