=LAMBDA(txt,
  LET(
    asc, ASCII_(txt),
    IF(asc = "",
  LET(
    modBig,  4294967296,
    maskAll, 4294967295,

    addM, LAMBDA(xa, ya, MOD(xa + ya, modBig)),
    notM, LAMBDA(xa, BITXOR(xa, maskAll)),
    rotR, LAMBDA(xa, nb,
      MOD(
        BITOR(
          BITRSHIFT(xa, nb),
          BITLSHIFT(BITAND(xa, 2^nb - 1), 32 - nb)
        ),
        modBig
      )
    ),
    shR, LAMBDA(xa, nb, BITRSHIFT(xa, nb)),

    mlen, LEN(txt),
    plen, 64 * (INT((mlen + 8) / 64) + 1),
    nblk, plen / 64,

    padByte, LAMBDA(idx,
      IF(idx < mlen,
        CODE(MID(txt, idx + 1, 1)),
        IF(idx = mlen,
          128,
          IF(idx >= plen - 8,
            MOD(INT(mlen * 8 / 256^(plen - 1 - idx)), 256),
            0
          )
        )
      )
    ),

    getWord, LAMBDA(blkIdx, wordIdx,
      LET(
        pp, blkIdx * 64 + wordIdx * 4,
          padByte(pp)     * 16777216
        + padByte(pp + 1) * 65536
        + padByte(pp + 2) * 256
        + padByte(pp + 3)
      )
    ),

    karr,  SHA256K_(),
    hInit, SHA256I_(),

    bigS0, LAMBDA(xa, BITXOR(BITXOR(rotR(xa,  2), rotR(xa, 13)), rotR(xa, 22))),
    bigS1, LAMBDA(xa, BITXOR(BITXOR(rotR(xa,  6), rotR(xa, 11)), rotR(xa, 25))),
    smlS0, LAMBDA(xa, BITXOR(BITXOR(rotR(xa,  7), rotR(xa, 18)), shR(xa,  3))),
    smlS1, LAMBDA(xa, BITXOR(BITXOR(rotR(xa, 17), rotR(xa, 19)), shR(xa, 10))),

    chFn,  LAMBDA(xa, ya, za, BITXOR(BITAND(xa, ya), BITAND(notM(xa), za))),
    majFn, LAMBDA(xa, ya, za, BITXOR(BITXOR(BITAND(xa, ya), BITAND(xa, za)), BITAND(ya, za))),

    buildW, LAMBDA(blk,
      LET(
        initW, getWord(blk, SEQUENCE(1, 16, 0)),
        REDUCE(initW, SEQUENCE(48, , 16),
          LAMBDA(wA, jj,
            HSTACK(wA,
              addM(
                addM(
                  addM(
                    smlS1(INDEX(wA, 1, jj -  1)),
                          INDEX(wA, 1, jj -  6)
                  ),
                  smlS0(INDEX(wA, 1, jj - 14))
                ),
                INDEX(wA, 1, jj - 15)
              )
            )
          )
        )
      )
    ),

    finalH, REDUCE(hInit, SEQUENCE(nblk, , 0),
      LAMBDA(hh, blk,
        LET(
          wArr, buildW(blk),
          compr, REDUCE(hh, SEQUENCE(64, , 1),
            LAMBDA(state, ii,
              LET(
                aa, INDEX(state, 1, 1),
                bb, INDEX(state, 1, 2),
                cc, INDEX(state, 1, 3),
                dd, INDEX(state, 1, 4),
                ee, INDEX(state, 1, 5),
                ff, INDEX(state, 1, 6),
                gg, INDEX(state, 1, 7),
                hp, INDEX(state, 1, 8),
                tone, addM(addM(addM(addM(hp, bigS1(ee)), chFn(ee, ff, gg)), INDEX(karr, 1, ii)), INDEX(wArr, 1, ii)),
                ttwo, addM(bigS0(aa), majFn(aa, bb, cc)),
                CHOOSE({1,2,3,4,5,6,7,8},
                  addM(tone, ttwo), aa, bb, cc,
                  addM(dd, tone),   ee, ff, gg
                )
              )
            )
          ),
          SHA256A_(hh, compr)
        )
      )
    ),

    SHA256H_(finalH)
  ),
      asc
    )
  )
)
