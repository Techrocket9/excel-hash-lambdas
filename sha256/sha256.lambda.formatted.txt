=LAMBDA(txt,
  LET(
    modBig,  4294967296,
    maskAll, 4294967295,

    addM, LAMBDA(xa, ya, MOD(xa + ya, modBig)),
    notM, LAMBDA(xa, BITXOR(xa, maskAll)),
    rotR, LAMBDA(xa, nb,
      MOD(
        BITOR(
          BITRSHIFT(xa, nb),
          BITLSHIFT(BITAND(xa, 2^nb - 1), 32 - nb)
        ),
        modBig
      )
    ),
    shR, LAMBDA(xa, nb, BITRSHIFT(xa, nb)),

    mlen, LEN(txt),
    plen, 64 * (INT((mlen + 8) / 64) + 1),
    nblk, plen / 64,

    padByte, LAMBDA(idx,
      IF(idx < mlen,
        CODE(MID(txt, idx + 1, 1)),
        IF(idx = mlen,
          128,
          IF(idx >= plen - 8,
            MOD(INT(mlen * 8 / 256^(plen - 1 - idx)), 256),
            0
          )
        )
      )
    ),

    getWord, LAMBDA(blkIdx, wordIdx,
      LET(
        pp, blkIdx * 64 + wordIdx * 4,
          padByte(pp)     * 16777216
        + padByte(pp + 1) * 65536
        + padByte(pp + 2) * 256
        + padByte(pp + 3)
      )
    ),

    karr, {
      1116352408, 1899447441, 3049323471, 3921009573,  961987163, 1508970993, 2453635748, 2870763221,
      3624381080,  310598401,  607225278, 1426881987, 1925078388, 2162078206, 2614888103, 3248222580,
      3835390401, 4022224774,  264347078,  604807628,  770255983, 1249150122, 1555081692, 1996064986,
      2554220882, 2821834349, 2952996808, 3210313671, 3336571891, 3584528711,  113926993,  338241895,
       666307205,  773529912, 1294757372, 1396182291, 1695183700, 1986661051, 2177026350, 2456956037,
      2730485921, 2820302411, 3259730800, 3345764771, 3516065817, 3600352804, 4094571909,  275423344,
       430227734,  506948616,  659060556,  883997877,  958139571, 1322822218, 1537002063, 1747873779,
      1955562222, 2024104815, 2227730452, 2361852424, 2428436474, 2756734187, 3204031479, 3329325298
    },

    hInit, {
      1779033703, 3144134277, 1013904242, 2773480762,
      1359893119, 2600822924,  528734635, 1541459225
    },

    bigS0, LAMBDA(xa, BITXOR(BITXOR(rotR(xa,  2), rotR(xa, 13)), rotR(xa, 22))),
    bigS1, LAMBDA(xa, BITXOR(BITXOR(rotR(xa,  6), rotR(xa, 11)), rotR(xa, 25))),
    smlS0, LAMBDA(xa, BITXOR(BITXOR(rotR(xa,  7), rotR(xa, 18)), shR(xa,  3))),
    smlS1, LAMBDA(xa, BITXOR(BITXOR(rotR(xa, 17), rotR(xa, 19)), shR(xa, 10))),

    chFn,  LAMBDA(xa, ya, za, BITXOR(BITAND(xa, ya), BITAND(notM(xa), za))),
    majFn, LAMBDA(xa, ya, za, BITXOR(BITXOR(BITAND(xa, ya), BITAND(xa, za)), BITAND(ya, za))),

    buildW, LAMBDA(blk,
      LET(
        initW, getWord(blk, SEQUENCE(1, 16, 0)),
        REDUCE(initW, SEQUENCE(48, , 16),
          LAMBDA(wA, jj,
            HSTACK(wA,
              addM(
                addM(
                  addM(
                    smlS1(INDEX(wA, 1, jj -  1)),
                          INDEX(wA, 1, jj -  6)
                  ),
                  smlS0(INDEX(wA, 1, jj - 14))
                ),
                INDEX(wA, 1, jj - 15)
              )
            )
          )
        )
      )
    ),

    finalH, REDUCE(hInit, SEQUENCE(nblk, , 0),
      LAMBDA(hh, blk,
        LET(
          wArr, buildW(blk),
          compr, REDUCE(hh, SEQUENCE(64, , 1),
            LAMBDA(state, ii,
              LET(
                aa, INDEX(state, 1, 1),
                bb, INDEX(state, 1, 2),
                cc, INDEX(state, 1, 3),
                dd, INDEX(state, 1, 4),
                ee, INDEX(state, 1, 5),
                ff, INDEX(state, 1, 6),
                gg, INDEX(state, 1, 7),
                hp, INDEX(state, 1, 8),
                tone, addM(addM(addM(addM(hp, bigS1(ee)), chFn(ee, ff, gg)), INDEX(karr, 1, ii)), INDEX(wArr, 1, ii)),
                ttwo, addM(bigS0(aa), majFn(aa, bb, cc)),
                CHOOSE({1,2,3,4,5,6,7,8},
                  addM(tone, ttwo), aa, bb, cc,
                  addM(dd, tone),   ee, ff, gg
                )
              )
            )
          ),
          CHOOSE({1,2,3,4,5,6,7,8},
            addM(INDEX(hh, 1, 1), INDEX(compr, 1, 1)),
            addM(INDEX(hh, 1, 2), INDEX(compr, 1, 2)),
            addM(INDEX(hh, 1, 3), INDEX(compr, 1, 3)),
            addM(INDEX(hh, 1, 4), INDEX(compr, 1, 4)),
            addM(INDEX(hh, 1, 5), INDEX(compr, 1, 5)),
            addM(INDEX(hh, 1, 6), INDEX(compr, 1, 6)),
            addM(INDEX(hh, 1, 7), INDEX(compr, 1, 7)),
            addM(INDEX(hh, 1, 8), INDEX(compr, 1, 8))
          )
        )
      )
    ),

    hexByte, LAMBDA(bv, RIGHT("0" & DEC2HEX(bv), 2)),
    wordHex, LAMBDA(ww,
        hexByte(MOD(INT(ww / 16777216), 256))
      & hexByte(MOD(INT(ww / 65536),    256))
      & hexByte(MOD(INT(ww / 256),      256))
      & hexByte(MOD(ww,                 256))
    ),

    LOWER(
        wordHex(INDEX(finalH, 1, 1))
      & wordHex(INDEX(finalH, 1, 2))
      & wordHex(INDEX(finalH, 1, 3))
      & wordHex(INDEX(finalH, 1, 4))
      & wordHex(INDEX(finalH, 1, 5))
      & wordHex(INDEX(finalH, 1, 6))
      & wordHex(INDEX(finalH, 1, 7))
      & wordHex(INDEX(finalH, 1, 8))
    )
  )
)
