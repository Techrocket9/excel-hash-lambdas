=LAMBDA(tx,
  LET(
    n,LEN(tx),
    p,136*INT(n/136+1),
    i,SEQUENCE(p,,0),
    u,IFERROR(UNICODE(MID(tx,i+1,1)),999),
    by,IF(i<n,u,0)+(i=n)*6+(i=p-1)*128,
    k,SEQUENCE(25,,0),
    x,MOD(k,5),
    z,SEQUENCE(1,64,0),
    cm,--(MOD(TOROW(k),5)=SEQUENCE(5,,0)),
    pz,MOD(x+3*INT(k/5),5)+5*x+1,
    rt,MOD(z-{0;44;43;21;14;28;20;3;45;61;1;6;25;8;18;27;36;10;15;56;62;55;39;41;2},64)+1,
    ca,k-x+MOD(x+1,5)+1,
    cb,k-x+MOD(x+2,5)+1,
    F,LAMBDA(st,
      REDUCE(st,{1;26;94;112;31;33;121;85;14;12;53;38;63;79;93;83;82;72;22;102;121;88;33;116},LAMBDA(A,qc,
        LET(
          D,MMULT(cm,A),
          E,MOD(A+INDEX(D,MOD(x+4,5)+1,z+1)+INDEX(D,MOD(x+1,5)+1,MOD(z-1,64)+1),2),
          B,INDEX(E,pz,rt),
          MOD(B+(1-INDEX(B,ca,z+1))*INDEX(B,cb,z+1)+(k=0)*(BITAND(z,z+1)=0)*MOD(INT(qc/(z+1)),2),2)
        )
      ))
    ),
    S,REDUCE(k*z*0,SEQUENCE(p/136,,0),LAMBDA(st,bk,
      F(MOD(st+IF(k<17,MOD(INT(INDEX(by,136*bk+8*k+INT(z/8)+1)/2^MOD(z,8)),2),0),2))
    )),
    IF(OR((i<n)*(u>127)),"Error: non-ASCII input detected",LOWER(CONCAT(DEC2HEX(MMULT(WRAPROWS(TOCOL(TAKE(S,4)),8),2^SEQUENCE(8,,0)),2))))
  )
)
